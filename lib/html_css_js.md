### 代码运行结果题目
1. 以下代码运行结果是
```js

function test() {
    console.log(1);
    setTimeout(() => {
        console.log(2);
    }, 1000)
}
test();
setTimeout(() => {
    console.log(3);
})
new Promise((resolve) => {
    console.log(4);
    setTimeout(() => {
        console.log(5);
    }, 100)
    resolve()
}).then(() => {
    setTimeout(() => {
        console.log(6);
    }, 0);
    console.log(7);
})
console.log(8);
// 1 4 8 7 3 6 5 2

```

2、下面代码运行结果是：2 4 3 1
```js
setTimeout(() => {
    console.log(1);
}, 1000)

new Promise(function (resolve) {
    console.log(2);
    for (var i = 0; i < 10000; i++) {
        i == 99 && resolve()
    }
}).then(() => {
    console.log(3);
})
console.log(4);
```
